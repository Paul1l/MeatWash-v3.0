# MEATWASH — meatwash.ru
# Статический сайт. Apache на хостинге рег.ру.
#
# Это файл .htaccess корня сайта на хостинге: в репозитории он лежит как
# server/htaccess.txt, на хостинг загружается под именем .htaccess (вместе с
# server/404.html → 404.html). Порядок выкладки и проверки — README, «Домен meatwash.ru».
# Ссылки на сайте ведут на адреса без .html (/services): без этого файла все
# страницы, кроме главной, отвечают 404.

AddDefaultCharset UTF-8
# -MultiViews: адрес /services отдают только правила ниже. С MultiViews Apache
# сам подбирает services.html и пишет в ответ Content-Location: services.html.
# Если после заливки весь сайт отвечает 500 — хостинг не разрешает менять
# MultiViews: уберите из строки «-MultiViews», правила работают и без этого.
Options -Indexes -MultiViews
DirectoryIndex index.html

<IfModule mod_rewrite.c>
  RewriteEngine On

  # ── Адреса страниц без .html (решение 19.10) ──────────────────────────────
  # https://meatwash.ru/services отдаёт файл services.html. Старые адреса *.html
  # (закладки, поиск, внешние ссылки) — 301 на новые, параметры (?utm_…)
  # переносятся. Только для файлов в корне сайта: копию /v2/ (у неё свой
  # .htaccess) и файл Вебмастера yandex_….html правила не трогают.
  # Три переадресации ниже стоят до правил https/www: их адрес уже полный
  # https://meatwash.ru/…, поэтому http://, www и .html исправляются одним 301.
  # THE_REQUEST — строка запроса посетителя: внутренняя подстановка
  # /services → services.html и страница 404 её не меняют, поэтому
  # переадресации по кругу нет.

  # Главная: /index.html, /index, /index/, /index.html/ → /
  RewriteCond %{THE_REQUEST} ^[A-Z]+\ [^?\ ]*/index[^/?\ ]*/?[?\ ]
  RewriteRule ^index(\.html)?/?$ https://meatwash.ru/ [R=301,L]

  # /services.html → /services, если такой файл есть. Только исходный запрос
  # (REDIRECT_STATUS пуст) и только если посетитель сам написал .html
  # (в том числе //services.html и services%2Ehtml).
  RewriteCond %{ENV:REDIRECT_STATUS} ^$
  RewriteCond %{THE_REQUEST} ^[A-Z]+\ [^?\ ]*(\.|%2e)html[?\ ] [NC]
  RewriteCond $1 !^(yandex_|404$)
  RewriteCond %{REQUEST_FILENAME} -f
  RewriteRule ^([^/]+)\.html$ https://meatwash.ru/$1 [R=301,L]

  # /services/ и /services.html/ → /services, если есть services.html
  RewriteCond $1 !^(yandex_|404$)
  RewriteCond %{REQUEST_FILENAME} ^(.*?)(\.html)?$
  RewriteCond %1.html -f
  RewriteRule ^([^/.]+)(\.html)?/$ https://meatwash.ru/$1 [R=301,L]

  # Папка без слеша (/v2) → https://meatwash.ru/v2/ одним 301. Иначе Apache за
  # nginx сам отправляет на http://meatwash.ru/v2/, а оттуда снова на https.
  RewriteCond %{REQUEST_FILENAME} -d
  RewriteRule ^(.*[^/])$ https://meatwash.ru/$1/ [R=301,L]

  # ── http и www → https://meatwash.ru ───────────────────────────────────────
  # Включено 05.10.2026, сертификат GlobalSign до 16.04.2027. Если после заливки
  # браузер пишет «слишком много переадресаций» — верните .htaccess, скачанный
  # с сервера перед заменой. Путь берётся из строки запроса посетителя, а не из
  # REQUEST_URI: так /services не превращается в /services.html, даже если
  # хостинг включит MultiViews. Параметры (?…) в шаблон не берём — Apache допишет
  # их сам; «?» из подстановки Apache 2.4.60+ отклоняет с кодом 403.
  RewriteCond %{HTTPS} !=on
  RewriteCond %{HTTP:X-Forwarded-Proto} !https
  RewriteCond %{THE_REQUEST} ^[A-Z]+\ (/[^?\ ]*)
  RewriteRule ^ https://meatwash.ru%1 [R=301,L,NE]
  RewriteCond %{HTTP_HOST} ^www\.(.+)$ [NC]
  RewriteCond %{THE_REQUEST} ^[A-Z]+\ (/[^?\ ]*)
  RewriteRule ^ https://meatwash.ru%1 [R=301,L,NE]

  # /services → файл services.html (адрес в браузере не меняется). Стоит после
  # https/www: иначе http://…/services ушёл бы на https://…/services.html.
  RewriteCond $1 !^(yandex_|404$)
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteCond %{REQUEST_FILENAME}.html -f
  RewriteRule ^([^/.]+)$ /$1.html [L]
</IfModule>

ErrorDocument 404 /404.html

# Типы, которые старый Apache может не знать
<IfModule mod_mime.c>
  AddType image/webp                 .webp
  AddType image/svg+xml              .svg
  AddType font/woff2                 .woff2
  AddType font/ttf                   .ttf
  AddType model/gltf-binary          .glb
  AddType application/manifest+json  .webmanifest
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css text/plain text/xml
  AddOutputFilterByType DEFLATE application/javascript application/json application/xml image/svg+xml
</IfModule>

<IfModule mod_expires.c>
  ExpiresActive On
  # Страницы всегда перепроверяются: правки доходят сразу
  ExpiresByType text/html                 "access plus 0 seconds"
  ExpiresByType application/xml           "access plus 0 seconds"
  ExpiresByType text/css                  "access plus 1 day"
  ExpiresByType application/javascript    "access plus 1 day"
  ExpiresByType image/webp                "access plus 7 days"
  ExpiresByType image/png                 "access plus 7 days"
  ExpiresByType image/svg+xml             "access plus 7 days"
  ExpiresByType font/woff2                "access plus 30 days"
  ExpiresByType model/gltf-binary         "access plus 1 year"
</IfModule>

<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  <FilesMatch "\.html$">
    Header set Cache-Control "public, max-age=0, must-revalidate"
  </FilesMatch>
  # Имена моделей содержат хеш — их можно держать вечно
  <FilesMatch "\.[0-9a-f]{10}\.glb$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
</IfModule>
